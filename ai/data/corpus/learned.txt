Axen: ```python
def is_prime(n):
    """Check if a number is prime."""
    if n <= 1:
        return False
    if n <= 3:
        return True
    if n % 2 == 0 or n % 3 == 0:
        return False
    i = 5
    while i * i <= n:
        if n % i == 0 or n % (i + 2) == 0:
            return False
        i += 6
    return True

# Test cases
print(is_prime(2))  # True
print(is_prime(3))  # True
print(is_prime(4))  # False
```

User: write a python command-line to-do app that can add, list, complete, and delete tasks, and saves them to a JSON file
Axen: ```python
import json
import argparse

# Load tasks from JSON file
def load_tasks():
    try:
        with open('tasks.json', 'r') as file:
            return json.load(file)
    except FileNotFoundError:
        return []

# Save tasks to JSON file
def save_tasks(tasks):
    with open('tasks.json', 'w') as file:
        json.dump(tasks, file)

# Add a new task
def add_task(tasks, task):
    tasks.append(task)
    save_tasks(tasks)
    print(f"Added: {task}")

# List all tasks
def list_tasks(tasks):
    if not tasks:
        print("No tasks.")
    else:
        print("Tasks:")
        for index, task in enumerate(tasks, start=1):
            print(f"{index}. {task}")

# Complete a task
def complete_task(tasks, task):
    tasks.remove(task)
    save_tasks(tasks)
    print(f"Completed: {task}")

# Delete a task
def delete_task(tasks, task):
    tasks.remove(task)
    save_tasks(tasks)
    print(f"Deleted: {task}")

# Main command-line interface
def main():
    parser = argparse.ArgumentParser(description="A simple command-line to-do app.")
    parser.add_argument('command', choices=['add', 'list', 'complete', 'delete', 'save', 'load', 'exit'], help="Command to execute: add, list, complete, delete, save, load, or exit.")
    parser.add_argument('task', nargs='*', help="Task to add or complete, or task to delete.")

    args = parser.parse_args()

    if args.command == 'save':
        save_tasks(args.task)
    elif args.command == 'load':
        tasks = load_tasks()
    elif args.command == 'exit':
        print("Exiting the to-do app.")
        return
    else:
        tasks = load_tasks()

    if args.command == 'add':
        add_task(tasks, ' '.join(args.task))
    elif args.command == 'list':
        list_tasks(tasks)
    elif args.command == 'complete':
        complete_task(tasks, ' '.join(args.task))
    elif args.command == 'delete':
        delete_task(tasks, ' '.join(args.task))

if __name__ == '__main__':
    main()
```
